az appservice plan create \
    --name app01-plan \
    -g lab-compute-app01-rg \
    --sku B1 \
    --is-linux \
    -l eastus

az webapp create \
    --name app01-webapp-0987 \
    -g lab-compute-app01-rg \
    --plan app01-plan \
    --runtime "NODE|24-lts"

az webapp config appsettings set \
    -g lab-compute-app02-rg \
    -n dev-tool-app-2440614902 \
    --settings @dev-tool-app-2440614902-settings.json

az appservice plan update \
  -g lab-compute-app03-rg \
  -n dev-tool-plan \
  --number-of-workers 3

az webapp deployment slot create \
  -g lab-compute-app04-rg \
  -n dev-tool-app-1920023309 \
  --slot staging 

az webapp config appsettings set \
    -g lab-compute-app04-rg \
    -n dev-tool-app-1920023309 \
    --slot staging \
    --settings @dev-tool-app-1920023309-settings.json

az webapp config appsettings list \
  -n dev-tool-app-1920023309 \
  --slot staging \
  -g lab-compute-app04-rg

az webapp deployment slot swap \
  --slot staging \
  --name dev-tool-app-1920023309 \
  -g lab-compute-app04-rg
